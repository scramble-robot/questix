cmake_minimum_required(VERSION 3.16)
project(questix_pwm_guard)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
add_library(questix_pwm_guard_io src/sysfs.cpp)
target_include_directories(questix_pwm_guard_io PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
add_executable(questix_pwm_guard src/daemon.cpp)
target_link_libraries(questix_pwm_guard questix_pwm_guard_io)
add_executable(questix_pwm_ctl src/ctl.cpp)
target_link_libraries(questix_pwm_ctl questix_pwm_guard_io)
install(TARGETS questix_pwm_guard_io EXPORT guardTargets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(TARGETS questix_pwm_guard questix_pwm_ctl RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY systemd deploy DESTINATION share/${PROJECT_NAME})
ament_export_targets(guardTargets HAS_LIBRARY_TARGET)
ament_export_include_directories(include)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_guard_core test/test_core.cpp)
  target_link_libraries(test_guard_core questix_pwm_guard_io)
  ament_add_gtest(test_guard_sysfs test/test_sysfs.cpp)
  target_link_libraries(test_guard_sysfs questix_pwm_guard_io)
  add_executable(questix_pwm_guard_test_server src/daemon.cpp)
  target_compile_definitions(questix_pwm_guard_test_server PRIVATE QUESTIX_TEST_FAKE_IO)
  target_link_libraries(questix_pwm_guard_test_server questix_pwm_guard_io)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME test_manual_start COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_manual_start.py)
  add_test(NAME test_guard_ipc COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_ipc.py $<TARGET_FILE:questix_pwm_guard_test_server>)
  set_tests_properties(test_guard_ipc PROPERTIES TIMEOUT 30)
endif()
ament_package()
